cmake_minimum_required(VERSION 3.26)
project(shorecrab_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LLAMA_CPP_SOURCE)
  message(FATAL_ERROR "Set LLAMA_CPP_SOURCE to the pinned llama.cpp checkout")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_CPP_SOURCE} llama)
target_sources(llama PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp)
target_include_directories(llama PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(llama-shorecrab cli.cpp)
target_link_libraries(llama-shorecrab PRIVATE llama)
